# Compute the maximum of three numbers
    int a;
    int b;
    int c;
    int max;

    read(a);
    read(b);
    read(c);

    if (a >= b) {
        if (a >= c) {
            max = a;
        } else {
            max = c;
        }
    } else {
        if (b >= c) {
            max = b;
        } else {
            max = c;
        }
    }

    print(max);
   }
